const realAsset = name => new URL(`./realistic/${name}`, import.meta.url).href;
const realVertex = `
  varying vec3 vWorld;
  varying vec3 vNormal;
  varying vec2 vUv;
  void main(){
    vUv=uv;
    vec4 w=modelMatrix*vec4(position,1.0);
    vWorld=w.xyz; vNormal=normalize(mat3(modelMatrix)*normal);
    gl_Position=projectionMatrix*viewMatrix*w;
  }
`;
const realNoise = `
  float hash(vec3 p){p=fract(p*.1031);p+=dot(p,p.yzx+33.33);return fract((p.x+p.y)*p.z);}
  float noise(vec3 p){
    vec3 i=floor(p),f=fract(p);f=f*f*(3.0-2.0*f);
    return mix(mix(mix(hash(i),hash(i+vec3(1,0,0)),f.x),mix(hash(i+vec3(0,1,0)),hash(i+vec3(1,1,0)),f.x),f.y),
      mix(mix(hash(i+vec3(0,0,1)),hash(i+vec3(1,0,1)),f.x),mix(hash(i+vec3(0,1,1)),hash(i+vec3(1,1,1)),f.x),f.y),f.z);
  }
  float fbm(vec3 p){return noise(p)*.57+noise(p*2.03)*.28+noise(p*4.07)*.15;}
`;
const realSkyFragment = `
  varying vec3 vWorld; varying vec3 vNormal; varying vec2 vUv;
  uniform vec3 uCam,uSun; uniform float uTime,uWarm,uMotion;
  uniform sampler2D uSky;uniform float uSkyReady;
  ${realNoise}
  vec3 daylight(vec3 d){
    float horizon=pow(1.0-abs(d.y),3.0);
    vec3 top=mix(vec3(.115,.265,.47),vec3(.055,.105,.18),uWarm);
    vec3 edge=mix(vec3(.63,.72,.79),vec3(.78,.48,.24),uWarm);
    vec3 col=mix(top,edge,horizon);
    float sunDot=max(0.0,dot(d,uSun));
    col+=vec3(1.0,.75,.42)*pow(sunDot,180.0)*.75;
    col+=vec3(1.0,.84,.59)*pow(sunDot,2400.0)*6.0;
    return col;
  }
  void main(){
    vec3 rd=normalize(vWorld-uCam);
    float air=1.0-smoothstep(170.0,600.0,uCam.y);
    float star=step(.99945,hash(floor(rd*1700.0)))*pow(max(.0,1.0-air),2.0);
    vec3 col=mix(vec3(.0015,.003,.009)+star*vec3(.62,.71,.86),daylight(rd),air);
    if(uSkyReady>.5){vec2 skyUv=vec2(.5+atan(rd.z,rd.x)/6.283185,.5+asin(clamp(rd.y,-1.0,1.0))/3.141593);col=mix(col,texture2D(uSky,skyUv).rgb,air*.72);}
    // Ray-marched world-space cloud volume. No sprite clouds or flat cards.
    if(abs(rd.y)>.002 && uCam.y>1.0){
      float ta=(42.0-uCam.y)/rd.y,tb=(180.0-uCam.y)/rd.y;
      float begin=max(0.0,min(ta,tb)),finish=max(ta,tb);
      float lengthRay=min(700.0,max(0.0,finish-begin));
      float stepSize=lengthRay/18.0,trans=1.0;
      vec3 accumulated=vec3(0.0);
      for(int i=0;i<18;i++){
        vec3 p=uCam+rd*(begin+(float(i)+.5)*stepSize);
        float heightMask=smoothstep(42.0,68.0,p.y)*(1.0-smoothstep(126.0,180.0,p.y));
        float density=max(0.0,fbm(p*.021+vec3(uTime*.006*uMotion,0,0))-.44)*heightMask*3.0;
        float opacity=1.0-exp(-density*stepSize*.07);
        float lit=clamp(.72-density*.32+dot(rd,uSun)*.12,.35,1.0);
        vec3 cloud=mix(vec3(.62,.69,.77),vec3(1.0,.79,.57),uWarm*.65)*lit;
        accumulated+=cloud*opacity*trans;trans*=1.0-opacity;
      }
      col=col*trans+accumulated;
    }
    float deep=1.0-smoothstep(-3.0,1.0,uCam.y);
    float shaft=pow(max(0.0,dot(rd,normalize(vec3(.15,1,.08)))),11.0);
    vec3 underwater=vec3(.003,.027,.041)+vec3(.022,.105,.125)*shaft;
    col=mix(col,underwater,deep);
    gl_FragColor=vec4(col,1.0);
    #include <tonemapping_fragment>
    #include <colorspace_fragment>
  }
`;
const realEarthFragment = `
  varying vec3 vWorld; varying vec3 vNormal; varying vec2 vUv;
  uniform sampler2D uEarth,uSpecular;
  uniform vec3 uSun,uCam; uniform float uFade;
  void main(){
    vec3 n=normalize(vNormal),view=normalize(uCam-vWorld);
    vec3 day=texture2D(uEarth,vUv).rgb;
    float light=max(.0,dot(n,uSun));
    float ocean=texture2D(uSpecular,vUv).r;
    float spec=pow(max(0.0,dot(reflect(-uSun,n),view)),42.0)*ocean;
    float rim=pow(1.0-max(0.0,dot(n,view)),3.0);
    vec3 col=day*(.075+light*.95)+spec*vec3(.9,.84,.72)*.45;
    col+=vec3(.09,.28,.48)*rim*(.3+light);
    gl_FragColor=vec4(col,uFade);
    #include <tonemapping_fragment>
    #include <colorspace_fragment>
  }
`;
const realAtmosphereFragment = `

  void main(){

`;
const realOceanVertex = `
  uniform float uTime,uMotion;
  varying vec3 vWorld; varying vec3 vNormal; varying vec2 vUv;
  float wave(vec2 p){
    float t=uTime*uMotion;
    return sin(dot(p,vec2(.18,.11))+t*.82)*.21
      +sin(dot(p,vec2(-.27,.19))+t*1.13)*.095
      +sin(dot(p,vec2(.67,-.31))+t*1.37)*.025;
  }
  void main(){
    vec3 p=position;p.z=wave(p.xy);vUv=uv;
    vWorld=(modelMatrix*vec4(p,1.0)).xyz;
    float dx=(wave(p.xy+vec2(.08,0))-wave(p.xy-vec2(.08,0)))/.16;
    float dy=(wave(p.xy+vec2(0,.08))-wave(p.xy-vec2(0,.08)))/.16;
    vNormal=normalize(mat3(modelMatrix)*vec3(-dx,-dy,1.0));
    gl_Position=projectionMatrix*viewMatrix*vec4(vWorld,1.0);
  }
`;
const realOceanFragment = `
  uniform vec3 uCam,uSun;uniform float uTime,uWarm,uMotion;
  varying vec3 vWorld,vNormal;varying vec2 vUv;
  ${realNoise}
  float coast(float x){return 26.0+6.0*sin(x*.021+1.0)+3.0*sin(x*.057+2.3)+1.2*sin(x*.19+.7)+40.0*(sin(x*.0045+.4)-sin(.4))+16.0*exp(-pow((x-44.0)/9.0,2.0));}
  void main(){
    vec3 n=normalize(vNormal),v=normalize(uCam-vWorld);
    vec3 fine=vec3(noise(vec3(vWorld.xz*2.6,uTime*.04*uMotion))-.5,0,noise(vec3(vWorld.zx*3.1,uTime*.03*uMotion))-.5);
    n=normalize(n+fine*.11);
    vec3 refl=reflect(-v,n);
    float fresnel=.02+.98*pow(1.0-max(.0,dot(n,v)),5.0);
    vec3 sky=mix(vec3(.09,.20,.30),vec3(.76,.48,.25),pow(1.0-abs(refl.y),3.0)*uWarm);
    vec3 water=mix(vec3(.008,.080,.100),sky,fresnel);
    vec3 halfDir=normalize(v+uSun);
    float rough=.11,alpha=rough*rough,nDotH=max(.0,dot(n,halfDir));
    float D=alpha*alpha/(3.14159*pow(nDotH*nDotH*(alpha*alpha-1.0)+1.0,2.0));
    water+=vec3(1.0,.73,.42)*min(D*.014,4.0)*max(.0,dot(n,uSun));
    float shore=coast(vWorld.x)-vWorld.z;
    float foam=(1.0-smoothstep(0.0,2.4,shore))*smoothstep(-1.0,.1,shore);
    foam*=smoothstep(.38,.72,noise(vec3(vWorld.xz*3.5,uTime*.22*uMotion)));
    water=mix(water,vec3(.71,.76,.70),foam*.52);
    float underwater=1.0-smoothstep(-1.0,1.0,uCam.y);
    water=mix(water,vec3(.01,.11,.15),underwater*.8);
    float dist=length(uCam-vWorld);
    float fog=1.0-exp(-dist*.0028);
    water=mix(water,vec3(.40,.34,.27),fog*.5*(1.0-underwater));
    gl_FragColor=vec4(water,1.0);
    #include <tonemapping_fragment>
    #include <colorspace_fragment>
  }
`;

function useRealTextures(gl){
  const [maps,setMaps]=Y.useState({});
  Y.useEffect(()=>{
    let active=true;const live=[];
    const sources={earth:['earth-day.jpg','srgb'],specular:['earth-specular.jpg',''],
      siding:['cottage-siding.webp','srgb'],leaves:['live-oak-leaves.webp','srgb'],sky:['coastal-sky.webp','srgb']};
    Object.entries(sources).forEach(([key,[file,space]])=>{
      const image=new Image();image.onload=()=>{
        const texture=new AnchorTexture(image);texture.colorSpace=space;texture.needsUpdate=true;
        texture.anisotropy=Math.min(4,gl.capabilities.getMaxAnisotropy());
        if(key==='siding'){texture.wrapS=texture.wrapT=1000;texture.repeat.set(2,1.8);}
        if(key==='sky')texture.mapping=303;
        live.push(texture);if(active)setMaps(prev=>({...prev,[key]:texture}));else texture.dispose();
      };image.src=realAsset(file);
    });
    return()=>{active=false;live.forEach(t=>t.dispose());};
  },[gl]);return maps;
}

  Y.useEffect(()=>{const old=scene.fog;scene.fog=fog;return()=>{scene.fog=old;};},[scene,fog]);

    c.jsx('hemisphereLight',{args:['#a6c1dc','#635342',1.25]}),
    c.jsx('directionalLight',{ref:sun,color:'#ffdfae',intensity:2.1,castShadow:true,
      'shadow-mapSize':[1024,1024],'shadow-camera-left':-100,'shadow-camera-right':100,

      const angle=random()*Math.PI*2,r=Math.sqrt(random())*3.8;
      add('leaf',x+Math.cos(angle)*r,1.5+height+(random()-.5)*2.1,z+Math.sin(angle)*r,2.1+random(),1.5+random(),1,
        (random()-.5)*1.5,random()*Math.PI,(random()-.5)*.65);
    }
  }
  return parts;
}
function RealCoastalTown({maps,small}){
  const group=Y.useRef(null);
  const batches=Y.useMemo(()=>{
    const data=realTownData(small),box=new _h(1,1,1),roof=realRoofGeometry(),trunk=new bc(.5,.68,1,16),leaf=new Id(1,1);
    const materials={
      wall:new AnchorMaterial({color:'#e2e3dc',map:maps.siding||null,roughness:.84}),
      trim:new AnchorMaterial({color:'#efeee6',roughness:.65}),
      roof:new AnchorMaterial({color:'#5c696b',metalness:.35,roughness:.46,side:2}),
      wood:new AnchorMaterial({color:'#735d48',roughness:.87}),
      glass:new AnchorMaterial({color:'#496b74',metalness:.7,roughness:.09}),
      light:new AnchorMaterial({color:'#efc17e',emissive:'#ffc579',emissiveIntensity:.8,roughness:.65}),
      bark:new AnchorMaterial({color:'#514a36',roughness:1}),
      leaf:new AnchorMaterial({color:'#8aa77a',map:maps.leaves||null,alphaTest:.55,side:2,roughness:.92}),
      road:new AnchorMaterial({color:'#444747',roughness:.98}),
      sand:new AnchorMaterial({color:'#b2a38a',roughness:1})
    };
    const items=Object.entries(data).map(([kind,instances])=>{
      const geo=kind==='roof'?roof:kind==='bark'?trunk:kind==='leaf'?leaf:box;
      const mesh=new Ch(geo,materials[kind],instances.length);const transform=new Eh();
      instances.forEach((instance,index)=>{
        transform.position.set(...instance.p);transform.rotation.set(...instance.r);transform.scale.set(...instance.s);transform.updateMatrix();mesh.setMatrixAt(index,transform.matrix);
      });mesh.instanceMatrix.needsUpdate=true;mesh.name=`real-coast-${kind}`;mesh.castShadow=kind!=='light';mesh.receiveShadow=true;
      if(kind==='leaf'&&!maps.leaves)mesh.visible=false;
      mesh.computeBoundingSphere();return mesh;
    });
    return {items,materials,geometries:[box,roof,trunk,leaf]};
  },[maps.siding,maps.leaves,small]);
  Y.useEffect(()=>()=>{batches.items.forEach(m=>m.dispose());Object.values(batches.materials).forEach(m=>m.dispose());batches.geometries.forEach(g=>g.dispose());},[batches]);
  ll(()=>{if(group.current)group.current.visible=Q.j>1.8;batches.materials.light.emissiveIntensity=.25+.85*q.warm;});
  return c.jsx('group',{ref:group,name:'realistic-coastal-neighborhood',children:batches.items.map(mesh=>c.jsx('primitive',{object:mesh},mesh.name))});
}
function RealSubmarineParticles({small}){
  const ref=Y.useRef(null);const geometry=Y.useMemo(()=>{
    const geometry=new Hd(),positions=[];for(let i=0;i<(small?100:260);i++){positions.push(Math.sin(i*7.43)*65,-5-(i%37),Math.cos(i*4.31)*75);}
    geometry.setAttribute('position',new AnchorFloat32(positions,3));return geometry;
  },[small]);
  Y.useEffect(()=>()=>geometry.dispose(),[geometry]);
  ll(()=>{if(ref.current){ref.current.visible=q.cam.y<1;ref.current.rotation.y=Q.reduced?0:q.time*.003;}});
  return c.jsx('points',{ref,geometry,children:c.jsx('pointsMaterial',{color:'#77a9b4',size:.045,transparent:true,opacity:.3,depthWrite:false})});
}
