// A connected mainland district. Dimensions are meters; sea level is y=0.
// The promenade remains an unobstructed corridor at z=34..46.
function districtData(small){
 const items=[];let seed=9142;const rand=()=>{seed=(Math.imul(seed,1664525)+1013904223)>>>0;return seed/4294967296;};
 const add=(kind,p,s,r=[0,0,0],color=null)=>items.push({kind,p,s,r,color});
 const box=(kind,x,y,z,w,h,d,color=null)=>add(kind,[x,y,z],[w,h,d],[0,0,0],color);
 const ground=2.8;
 function window(x,y,z,w,h){
  box('recess',x,y,z,w+.24,h+.24,.18);box('glass',x,y,z-.12,w,h,.045);
  box('trim',x-w/2-.06,y,z-.15,.10,h+.24,.12);box('trim',x+w/2+.06,y,z-.15,.10,h+.24,.12);
  box('trim',x,y+h/2+.06,z-.15,w+.22,.10,.12);box('trim',x,y-h/2-.06,z-.22,w+.38,.10,.32);
  box('metal',x,y,z-.17,.04,h,.06);
  if(rand()>.42)box('interior',x,y,z-.09,w*.9,h*.9,.03);
 }
 function terrace(x,y,z,w,d){
  box('concrete',x,y,z,w,.20,d);box('metal',x,y+1.05,z-d/2,.06,.06,.06);
  box('metal',x,y+1.05,z-d/2,w,.055,.055);
  for(let i=0;i<=Math.floor(w/.42);i++)box('metal',x-w/2+i*w/Math.floor(w/.42),y+.52,z-d/2,.027,1.04,.027);
  for(const side of [-1,1]){box('metal',x+side*w/2,y+1.05,z,.055,.055,d);for(let i=0;i<5;i++)box('metal',x+side*w/2,y+.52,z-d/2+i*d/4,.027,1.04,.027);}
 }
 function building(x,z,w,d,floors,color,style){
  const h=floors*3.4,front=z-d/2;
  box(style===1?'brick':'stucco',x,ground+h/2,z,w,h,d,color);
  box('stone',x,ground+.35,z,w+.15,.7,d+.15);
  // Cornice, parapet and shadowed roof recess, rather than one closed block.
  box('trim',x,ground+h+.06,z,w+.4,.28,d+.4);
  box('roof',x,ground+h+.23,z,w-.45,.07,d-.45);
  for(const side of [-1,1])box('stucco',x+side*(w/2-.09),ground+h+.50,z,.18,.75,d,color);
  box('stucco',x,ground+h+.5,z+d/2-.09,w,.75,.18,color);
  box('stucco',x,ground+h+.5,front+.09,w,.75,.18,color);
  box('metal',x+w*.23,ground+h+.65,z+.8,1.3,.75,1.3);
  const bays=Math.max(2,Math.floor(w/3.2));
  for(let f=0;f<floors;f++){
   const y=ground+f*3.4+1.85;
   const facadeKind=(f===0?'shop':'upper')+(style===1?'Brick':'White');
   for(const panel of [-1,1])add(facadeKind,[x+panel*w/4,ground+f*3.4+1.7,front-.18],[w/2,3.4,1],[0,Math.PI,0]);
   for(const side of [-1,1])for(const panel of [-1,1])add(facadeKind,[x+side*(w/2+.03),ground+f*3.4+1.7,z+panel*d/4],[d/2,3.4,1],[0,side*Math.PI/2,0]);
   if(f>0)box('trim',x,ground+f*3.4,front-.10,w+.15,.13,.18);
   for(let b=0;b<bays;b++){
    const wx=x+(b-(bays-1)/2)*w/bays;
    // Photographic glazing remains visible; balcony rails project from the facade.
    if(f>0&&style!==1&&b%2===0)terrace(wx,ground+f*3.4+.10,front-.95,2.3,1.7);
   }
  }
  // Ground floor canopies and timber doors establish human scale.
  box('wood',x,ground+1.35,front-.20,1.1,2.6,.12);
  for(const side of [-1,1])box('trim',x+side*w*.42,ground+h/2,front-.13,.23,h,.23);
  if(style!==2){add('awning',[x,ground+2.9,front-1],[w*.90,.09,1.9],[.12,0,0]);for(const side of [-1,1])box('metal',x+side*w*.4,ground+1.4,front-1.6,.07,2.8,.07);}
 }
 // A compact waterfront main street with varied facades and actual side streets.
 for(let row=0;row<(small?2:3);row++){
  const z=70+row*40;
  for(let col=0;col<(small?12:18);col++){
   const x=-205+col*25;
   if((x>-75&&x<-20&&row===0)||(x>-20&&x<10))continue;
   const colors=['#e4ded0','#adc0bb','#c7bda8','#f2e5c9','#b8c5c9','#d6bda8'];
   building(x,z,15+rand()*4,16+rand()*5,row===0?2+Math.floor(rand()*2):2+Math.floor(rand()*4),colors[(row+col)%colors.length],col%3);
  }
 }
 // Residential edge: deep shaded verandas, standing-seam roofs and shutters.
 for(let i=0;i<(small?7:12);i++){
  const x=95+i*23,z=70+(i%2)*37,w=10,d=12,h=5.5,front=z-d/2;
  box('siding',x,ground+h/2,z,w,h,d,['#ced9d5','#f0e6d1','#b8c4ca'][i%3]);
  add('gable',[x,ground+h,z],[w+1.1,3.4,d+1.5]);
  for(let seam=0;seam<12;seam++)for(const side of [-1,1])add('metal',[x+side*w*.25,ground+h+.79,z-d*.5+seam*(d+1)/11],[w*.53,.025,.035],[0,0,side*.30]);
  box('wood',x,ground+.18,front-2,w+1.6,.25,4.3);
  for(const sx of [-1,1]){box('trim',x+sx*w*.47,ground+2.5,front-3.8,.17,5,.17);window(x+sx*2.8,ground+2.8,front-.06,1.15,2.0);for(const side of [-1,1])box('shutter',x+sx*2.8+side*.85,ground+2.8,front-.16,.43,2.1,.07);}
  box('wood',x,ground+1.3,front-.08,1.0,2.6,.12);
  add('awning',[x,ground+5.1,front-2],[w+1.7,.10,4.5],[.11,0,0]);
  terrace(x,ground+.3,front-2,w+1.5,4.1);
 }
 // Sandy mainland, boulevard, raised sidewalks and waterfront promenade.
 box('asphalt',0,ground+.04,54,1250,.08,11);
 for(const z of [47,61])box('pavers',0,ground+.17,z,1250,.22,3.6);
 box('pavers',0,ground+.14,37,1250,.20,8);
 box('stone',0,1.2,27.5,1250,3.2,1.4);
 box('trim',0,ground+.28,28,1250,.28,2.2);
 for(let x=-620;x<620;x+=10)box('paint',x,ground+.09,54,4,.025,.13);
 for(const z of [48.5,59.5])box('paint',0,ground+.10,z,1240,.025,.12);
 for(const x of [-12,-87,88,238,-237]){
  box('asphalt',x,ground+.03,195,10,.06,275);
  for(const side of [-1,1])box('pavers',x+side*6.7,ground+.18,195,3.2,.25,275);
  for(let z=49;z<59;z+=1.4)box('paint',x,ground+.11,z,7.5,.03,.6);
 }
 // Central community square, colonnade and chapel facing the waterfront.
 box('pavers',-50,ground+.19,77,48,.28,34);
 building(-51,110,18,28,2,'#eee5d5',2);
 add('gable',[-51,ground+7,110],[20,5,30]);
 box('stucco',-51,ground+7,94,5.2,14,5.5,'#eee5d5');add('gable',[-51,ground+14,94],[6.3,5,6.3]);
 window(-51,ground+10,91.18,1.8,3.2);window(-51,ground+5.3,91.18,1.8,2.6);
 box('metal',-51,ground+17.4,94,.13,2.1,.13);box('metal',-51,ground+17.8,94,1.15,.13,.13);
 for(const x of [-61,-41])box('trim',x,ground+2.5,95,.24,5,.24);
 // Benches, lamps, planters, bollards and parked vehicles add inhabited scale.
 for(let x=-260;x<370;x+=22){
  box('metal',x,ground+2.6,31,.12,5.2,.12);box('metal',x,ground+5.25,31,.60,.12,.60);box('lamp',x,ground+5.05,31,.35,.4,.35);
  for(const z of [34,44])box('stone',x+5,ground+.35,z,.20,.7,.20);
  box('wood',x+8,ground+.52,43,2.2,.13,.6);box('wood',x+8,ground+1.0,43.25,2.2,.8,.08);
  for(const side of [-1,1])box('metal',x+8+side*.85,ground+.27,43,.1,.55,.55);
 }
 // A timber pier and slips remain above water and connect to the mainland.
 for(let z=-55;z<28;z+=.55)box('wood',-15,2.9,z,3.3,.15,.5);
 for(let z=-54;z<28;z+=5)for(const side of [-1,1]){box('wood',-15+side*1.4,.8,z,.22,4.7,.22);box('metal',-15+side*1.45,3.85,z,.05,1.9,.05);}
 for(const side of [-1,1])box('metal',-15+side*1.45,4.7,-13,.06,.06,80);
 // Palms use curved three-dimensional fronds with individual leaflets.
 const palms=[];for(let x=-270;x<400;x+=36)if(x < -110 || x > 55)palms.push([x,ground,45]);
 for(const x of [-69,-32])for(const z of [68,86])palms.push([x,ground,z]);
 for(let n=0;n<palms.length;n++){
  const [x,y,z]=palms[n],height=7+(n%4)*.7;
  add('trunk',[x+.3,y+height*.5,z],[.34,height,.34],[0,0,-.04]);
  for(let f=0;f<9;f++){
   const angle=f*Math.PI*2/9+n*.7;
   add('frond',[x+.6,y+height,z],[1,1,1],[0,angle,0]);
  }
 }
 return items;
}
function districtFrondGeometry(){
 const vertices=[],normals=[];const tri=(a,b,d)=>vertices.push(...a,...b,...d);
 // A curved central rib and paired tapered leaflets; no alpha cards.
 const spine=t=>[t*4,Math.sin(t*Math.PI)*1.1-t*t*.8,0];
 for(let k=0;k<17;k++){
  const t=.12+k*.046,p=spine(t),width=Math.sin(t*Math.PI)*.82;
  for(const side of [-1,1]){const tip=[p[0]+.52,p[1]-.24,side*width];tri([p[0]-.12,p[1],0],[p[0]+.13,p[1],0],tip);tri([p[0]-.12,p[1],0],tip,[p[0],p[1]-.045,side*.10]);}
 }
 const geo=new Hd();geo.setAttribute('position',new AnchorFloat32(vertices,3));geo.computeVertexNormals();return geo;
}
function CoastalDistrict({maps,small}){
 const ref=Y.useRef(null);
 const batches=Y.useMemo(()=>{
  const material=(color,extra={})=>new AnchorMaterial({color,roughness:.8,...extra});
  const mats={stucco:material('#ffffff',{map:maps.stucco,normalMap:maps.stuccoNormal}),brick:material('#ffffff',{map:maps.brick,normalMap:maps.brickNormal}),siding:material('#ffffff',{map:maps.wood}),stone:material('#aea391'),concrete:material('#c0b9aa'),trim:material('#eee8d9'),roof:material('#6c7978',{metalness:.3}),metal:material('#303b3b',{metalness:.75,roughness:.36}),wood:material('#c0a78a',{map:maps.wood}),glass:material('#556c70',{metalness:.85,roughness:.15}),recess:material('#292d2d'),interior:material('#8b7455',{emissive:'#c49a62',emissiveIntensity:.22}),awning:material('#a7b5af'),shutter:material('#36504c'),asphalt:material('#a0a0a0',{map:maps.asphalt}),pavers:material('#b6aaa0',{map:maps.pavers}),paint:material('#d6d1bb'),lamp:material('#e6cf9e',{emissive:'#ffd795',emissiveIntensity:1.8}),car:material('#ffffff',{metalness:.45,roughness:.28}),tire:material('#181b1b'),hull:material('#eee9dd'),trunk:material('#82715b'),frond:material('#426247',{side:2,roughness:.92})};
  for(const kind of ['stucco','brick','siding','wood','asphalt','pavers'])worldMappedMaterial(mats[kind],kind==='stucco'?.65:kind==='asphalt'?.12:.3);
  const atlasMaterials={};for(const [kind,offset] of Object.entries({shopWhite:[0,.5],shopBrick:[.5,.5],upperWhite:[0,0],upperBrick:[.5,0]})){const map=maps.facade?.clone();if(map){map.repeat.set(.5,.5);map.offset.set(...offset);map.needsUpdate=true;}atlasMaterials[kind]=material('#ffffff',{map,roughness:.75});mats[kind]=atlasMaterials[kind];}
  const facadePlane=new Id(1,1);
  const box=new _h(1,1,1),gable=realRoofGeometry(),cylinder=new bc(.5,.5,1,12),palmTrunk=new bc(.32,.60,1,20),sphere=new bc(.5,.5,1,16),leaf=districtFrondGeometry();
  const groups={};for(const item of districtData(small))(groups[item.kind]??=[]).push(item);
  const meshes=Object.entries(groups).map(([kind,parts])=>{
   const geo=kind.startsWith('shop')||kind.startsWith('upper')?facadePlane:kind==='gable'?gable:kind==='frond'?leaf:kind==='trunk'?palmTrunk:kind==='tire'?cylinder:kind==='hull'?new _h(1,1,1):box;
   const mesh=new Ch(geo,mats[kind==='gable'?'roof':kind],parts.length),obj=new Eh();
   parts.forEach((p,i)=>{obj.position.set(...p.p);obj.scale.set(...p.s);obj.rotation.set(...p.r);obj.updateMatrix();mesh.setMatrixAt(i,obj.matrix);if(p.color)mesh.setColorAt(i,new AnchorColor(p.color));});
   mesh.instanceMatrix.needsUpdate=true;if(mesh.instanceColor)mesh.instanceColor.needsUpdate=true;mesh.castShadow=!['paint','interior','glass','lamp','asphalt','pavers'].includes(kind);mesh.receiveShadow=true;mesh.name='district-'+kind;mesh.computeBoundingSphere();return mesh;
  });
  return {meshes,mats,geometries:[box,gable,cylinder,palmTrunk,sphere,leaf,facadePlane]};
 },[maps.stucco,maps.stuccoNormal,maps.brick,maps.brickNormal,maps.wood,maps.asphalt,maps.pavers,maps.facade,small]);
 Y.useEffect(()=>()=>{batches.meshes.forEach(m=>m.dispose());Object.values(batches.mats).forEach(m=>m.dispose());batches.geometries.forEach(g=>g.dispose());for(const k of ['shopWhite','shopBrick','upperWhite','upperBrick'])batches.mats[k]?.map?.dispose();},[batches]);
 ll(()=>{if(ref.current)ref.current.visible=q.cam.y<240;});
 return c.jsx('group',{ref,name:'mainland-coastal-community',children:batches.meshes.map(mesh=>c.jsx('primitive',{object:mesh},mesh.name))});
}
